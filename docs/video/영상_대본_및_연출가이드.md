```bash
cd docs/video/tools
python tts.py                 # 내레이션(장면별 길이 검사)
# PowerShell: export_slides.ps1 <pptx 경로> <작업폴더>\slides

python check_final.py         # 길이·스트림·주요 프레임 확인
```
